Private Const RUTA_WORD As String = "C:\Users\1025885461\Desktop\jhonny amorocho\aplicativos\informe general de solicitudes\Informe_General.docx"
Private Const CARPETA_INICIAL As String = "C:\Ruta\"

Public Const HOJA_DESTINO  As String = "Gráfico original"
Public Const HOJA_TABLA    As String = "Tabla general"
Public Const HOJA_RESUMEN  As String = "Resumen general"
Public Const NOMBRE_TABLA  As String = "TablaGeneral"

Private Const FILTRAR_SOLO_GESTANTES As Boolean = False

Private Const WORD_VISIBLE As Boolean = False

Private Const ACTUALIZAR_FECHA_CORTE As Boolean = False

Private mWbOrigen As Workbook
Private mWdApp As Object
Private mWdDoc As Object
Private mFilasExport As Long

Public Sub ActualizarInformeGestantes()
    Dim rutaExport As Variant
    Dim nFilas As Long
    Dim paso As String
    Dim msgWord As String
    Dim calcPrevio As Long

    calcPrevio = Application.Calculation
    On Error GoTo Manejador

    paso = "validación de la ruta del Word"
    If Len(Dir(RUTA_WORD)) = 0 Then
        MsgBox "No se encontró el Word en:" & vbCrLf & RUTA_WORD & vbCrLf & vbCrLf & _
               "Corrige la constante RUTA_WORD al inicio del módulo.", vbCritical, "Ruta no válida"
        Exit Sub
    End If

    paso = "selección del archivo exportado"
    If Len(Dir(CARPETA_INICIAL, vbDirectory)) > 0 Then
        On Error Resume Next
        ChDir CARPETA_INICIAL
        On Error GoTo Manejador
    End If

    rutaExport = Application.GetOpenFilename(FileFilter:="Archivos de Excel (*.xlsx),*.xlsx", Title:="Seleccione el archivo exportado del SAC")
    If VarType(rutaExport) = vbBoolean Then
        MsgBox "Proceso cancelado. No se modificó ningún archivo.", vbInformation
        Exit Sub
    End If

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    paso = "PASO 1 (carga del exportado)"
    nFilas = CargarDatos(CStr(rutaExport))
    If nFilas = 0 Then
        MsgBox "El archivo seleccionado no contiene filas de datos válidas.", vbExclamation, "Sin datos"
        GoTo Limpieza
    End If

    paso = "PASO 2 (ajuste de tabla y recálculo)"
    AjustarTabla nFilas

    If ACTUALIZAR_FECHA_CORTE Then
        ThisWorkbook.Worksheets(HOJA_RESUMEN).Range(CELDA_FECHA_CORTE).Value = Date
    End If

    Application.Calculation = xlCalculationAutomatic
    Application.CalculateFullRebuild
    ThisWorkbook.RefreshAll
    Application.CalculateUntilAsyncQueriesDone
    Application.Calculate

    paso = "PASO 2.5 (actualización del Dashboard)"
    ActualizarDashboardVBA
    ThisWorkbook.Save

    paso = "PASO 3 (actualización del Word)"
    msgWord = ActualizarWord(nFilas)

    MsgBox "Informe actualizado correctamente." & vbCrLf & vbCrLf & _
           "Filas en el exportado: " & mFilasExport & vbCrLf & _
           "Registros cargados: " & nFilas & vbCrLf & msgWord, vbInformation, "Proceso completado"

Limpieza:
    On Error Resume Next
    If Not mWbOrigen Is Nothing Then mWbOrigen.Close SaveChanges:=False
    Set mWbOrigen = Nothing
    If Not mWdDoc Is Nothing Then mWdDoc.Close 0
    Set mWdDoc = Nothing
    If Not mWdApp Is Nothing Then mWdApp.Quit 0
    Set mWdApp = Nothing

    Application.CutCopyMode = False
    Application.Calculation = calcPrevio
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    Exit Sub

Manejador:
    MsgBox "Error en " & paso & ":" & vbCrLf & vbCrLf & _
           "[" & Err.Number & "] " & Err.Description, vbCritical, "No se pudo completar"
    Resume Limpieza
End Sub

Private Function CargarDatos(ByVal ruta As String) As Long
    Dim wsO As Worksheet, wsD As Worksheet
    Dim ultFila As Long, ultCol As Long
    Dim rngTodo As Range

    Set mWbOrigen = Workbooks.Open(FileName:=ruta, ReadOnly:=True, UpdateLinks:=0)
    Set wsO = mWbOrigen.Worksheets(1)
    If wsO.AutoFilterMode Then wsO.AutoFilterMode = False

    ultFila = UltimaFila(wsO)
    ultCol = UltimaColumna(wsO)
    mFilasExport = ultFila - 1
    If ultFila < 2 Then GoTo Salir

    Set rngTodo = wsO.Range(wsO.Cells(1, 1), wsO.Cells(ultFila, ultCol))
    Set wsD = HojaPorPatron(ThisWorkbook, HOJA_DESTINO)
    wsD.Cells.ClearContents
    rngTodo.Copy Destination:=wsD.Range("A1")
    Application.CutCopyMode = False
    CargarDatos = ultFila - 1

Salir:
    If Not mWbOrigen Is Nothing Then mWbOrigen.Close SaveChanges:=False
    Set mWbOrigen = Nothing
End Function

Private Sub AjustarTabla(ByVal n As Long)
    Dim ws As Worksheet, lo As ListObject, actual As Long
    Set ws = ThisWorkbook.Worksheets(HOJA_TABLA)
    If Not ExisteTabla(ws, NOMBRE_TABLA) Then Exit Sub
    Set lo = ws.ListObjects(NOMBRE_TABLA)

    On Error Resume Next
    If lo.ShowAutoFilter Then
        If lo.AutoFilter.FilterMode Then lo.AutoFilter.ShowAllData
    End If
    On Error GoTo 0

    actual = lo.ListRows.Count
    If n > actual Then
        lo.Resize lo.Range.Resize(n + 1)
    ElseIf n < actual Then
        If Not lo.DataBodyRange Is Nothing Then
            lo.DataBodyRange.Rows(n + 1).Resize(actual - n).Delete Shift:=xlUp
        End If
    End If

    If n > 1 And Not lo.DataBodyRange Is Nothing Then
        lo.DataBodyRange.FillDown
    End If
End Sub

Private Sub ActualizarDashboardVBA()
    Dim ws As Worksheet
    Dim lo As ListObject
    Dim total As Long
    Dim sumaEdad As Double, nEdad As Long
    Dim arr As Variant
    Dim i As Long

    Set ws = ThisWorkbook.Worksheets(HOJA_RESUMEN)
    Set lo = ThisWorkbook.Worksheets(HOJA_TABLA).ListObjects(NOMBRE_TABLA)
    If lo Is Nothing Or lo.DataBodyRange Is Nothing Then Exit Sub

    total = lo.ListRows.Count

    ' --- tarjetas superiores ---
    ws.Range("B6").Value = total
    ws.Range("D6").Value = CountField(lo, "prioridad_caso", "Priorizado") + CountField(lo, "prioridad_caso", "Riesgo Vital")
    ws.Range("G6").Value = CountBarreras(lo)
    ws.Range("I6").Value = CountField(lo, "estadosolicitud", "Solucionada")
    ws.Range("D10").Value = CountNoSolucionadas(lo)

    sumaEdad = 0: nEdad = 0
    arr = lo.ListColumns("edad").DataBodyRange.Value
    If IsArray(arr) Then
        For i = 1 To UBound(arr, 1)
            If IsNumeric(arr(i, 1)) Then
                sumaEdad = sumaEdad + CDbl(arr(i, 1))
                nEdad = nEdad + 1
            End If
        Next i
    Else
        If IsNumeric(arr) Then
            sumaEdad = CDbl(arr)
            nEdad = 1
        End If
    End If
    If nEdad > 0 Then ws.Range("B10").Value = Round(sumaEdad / nEdad, 1) Else ws.Range("B10").Value = 0

    ' --- limpiar bloques del dashboard de forma segura ---
    ClearRange ws, 15, 105, 2, 11

    ' --- resumen general en bloques no solapados ---
    WriteCategoryBlock ws, 15, 2, 3, "prioridad_caso", Array("Priorizado", "Riesgo Vital", "Sin dato", "N/A")
    WriteCategoryBlock ws, 15, 7, 8, "grupo_etario", Array("Sin dato", "N/A")
    WriteCategoryBlock ws, 24, 2, 3, "barrera_acceso", Array("N/A", "Sin dato")
    WriteCategoryBlock ws, 24, 7, 8, "sexo", Array("Sin dato", "N/A")
    WriteCategoryBlock ws, 36, 2, 3, "estadosolicitud", Array("Solucionada", "Sin dato", "N/A")
    WriteCategoryBlock ws, 47, 2, 3, "solicitud", Array("Sin dato", "N/A")
    WriteCategoryBlock ws, 57, 7, 8, "subclasificacion", Array("Sin dato", "N/A")

    WriteEPSBlock ws, 36, 7, 8

    ' --- cabeceras y textos de apoyo ---
    ws.Range("B13").Value = "Riesgos de atención"
    ws.Range("G13").Value = "Distribución por grupo etario"
    ws.Range("B23").Value = "Barreras de acceso"
    ws.Range("G23").Value = "Participación por sexo"
    ws.Range("B35").Value = "Estado de la solicitud"
    ws.Range("G35").Value = "EPS"
    ws.Range("B46").Value = "Tipos de solicitud"
    ws.Range("G46").Value = "Subclasificaciones"

    ws.Range("K15").Value = "Observaciones"
End Sub

Private Sub WriteCategoryBlock(ws As Worksheet, ByVal startRow As Long, ByVal catCol As Long, ByVal valCol As Long, ByVal fieldName As String, ByVal excludeValues As Variant)
    Dim lo As ListObject
    Dim dict As Object
    Dim arr As Variant
    Dim i As Long, n As Long
    Dim key As Variant
    Dim keys As Variant, values As Variant
    Dim r As Long

    Set lo = ThisWorkbook.Worksheets(HOJA_TABLA).ListObjects(NOMBRE_TABLA)
    Set dict = CreateObject("Scripting.Dictionary")
    arr = lo.ListColumns(fieldName).DataBodyRange.Value

    If IsArray(arr) Then
        For i = 1 To UBound(arr, 1)
            key = Trim$(CStr(arr(i, 1)))
            If Len(key) > 0 And Not IsExcluded(key, excludeValues) Then
                dict(key) = dict(key) + 1
            End If
        Next i
    Else
        key = Trim$(CStr(arr))
        If Len(key) > 0 And Not IsExcluded(key, excludeValues) Then dict(key) = dict(key) + 1
    End If

    If dict.Count = 0 Then Exit Sub

    keys = dict.Keys
    values = dict.Items
    For i = LBound(values) To UBound(values) - 1
        For n = i + 1 To UBound(values)
            If CLng(values(n)) > CLng(values(i)) Then
                SwapVariant keys, i, n
                SwapVariant values, i, n
            End If
        Next n
    Next i

    r = startRow
    For i = LBound(keys) To UBound(keys)
        If r > startRow + 30 Then Exit For
        ws.Cells(r, catCol).Value = keys(i)
        ws.Cells(r, valCol).Value = values(i)
        r = r + 1
    Next i
End Sub

Private Sub WriteEPSBlock(ws As Worksheet, ByVal startRow As Long, ByVal epsCol As Long, ByVal valCol As Long)

    Dim i As Long, r As Long

    Set dict = CreateObject("Scripting.Dictionary")

        For i = 1 To UBound(arr, 1)
            key = Trim$(CStr(arr(i, 1)))
            If Len(key) > 0 Then dict(key) = dict(key) + 1
        Next i
    Else
        key = Trim$(CStr(arr))
        If Len(key) > 0 Then dict(key) = dict(key) + 1
    End If

    If dict.Count = 0 Then Exit Sub

    keys = dict.Keys
    values = dict.Items
    For i = LBound(values) To UBound(values) - 1
        For r = i + 1 To UBound(values)
            If CLng(values(r)) > CLng(values(i)) Then
                SwapVariant keys, i, r
                SwapVariant values, i, r
            End If
        Next r
    Next i

    r = startRow
    For i = LBound(keys) To UBound(keys)
        If r > startRow + 40 Then Exit For
        ws.Cells(r, epsCol).Value = keys(i)
        ws.Cells(r, valCol).Value = values(i)
        r = r + 1
    Next i
End Sub

Private Function CountField(lo As ListObject, ByVal fieldName As String, ByVal targetValue As String) As Long
    Dim arr As Variant, i As Long, total As Long
    Dim v As String
    arr = lo.ListColumns(fieldName).DataBodyRange.Value
    total = 0
    If IsArray(arr) Then
        For i = 1 To UBound(arr, 1)
            v = Trim$(CStr(arr(i, 1)))
            If LCase$(v) = LCase$(targetValue) Then total = total + 1
        Next i
    Else
        v = Trim$(CStr(arr))
        If LCase$(v) = LCase$(targetValue) Then total = 1
    End If
    CountField = total
End Function

Private Function CountBarreras(lo As ListObject) As Long
    Dim arr As Variant, i As Long, total As Long
    Dim v As String
    arr = lo.ListColumns("barrera_acceso").DataBodyRange.Value
    total = 0
    If IsArray(arr) Then
        For i = 1 To UBound(arr, 1)
            v = Trim$(CStr(arr(i, 1)))
            If Len(v) > 0 And LCase$(v) <> "n/a" And LCase$(v) <> "sin dato" Then total = total + 1
        Next i
    Else
        v = Trim$(CStr(arr))
        If Len(v) > 0 And LCase$(v) <> "n/a" And LCase$(v) <> "sin dato" Then total = 1
    End If
    CountBarreras = total
End Function

Private Function CountNoSolucionadas(lo As ListObject) As Long
    Dim arr As Variant, i As Long, total As Long
    Dim v As String
    arr = lo.ListColumns("estadosolicitud").DataBodyRange.Value
    total = 0
    If IsArray(arr) Then
        For i = 1 To UBound(arr, 1)
            v = Trim$(CStr(arr(i, 1)))
            If Len(v) > 0 And LCase$(v) <> "solucionada" Then total = total + 1
        Next i
    Else
        v = Trim$(CStr(arr))
        If Len(v) > 0 And LCase$(v) <> "solucionada" Then total = 1
    End If
    CountNoSolucionadas = total
End Function

Private Function IsExcluded(value As String, arr As Variant) As Boolean
    Dim i As Long
    If IsArray(arr) Then
        For i = LBound(arr) To UBound(arr)
            If LCase$(Trim$(CStr(arr(i)))) = LCase$(Trim$(value)) Then
                IsExcluded = True
                Exit Function
            End If
        Next i
    End If
    IsExcluded = False
End Function

Private Sub ClearRange(ws As Worksheet, ByVal startRow As Long, ByVal endRow As Long, ByVal startCol As Long, ByVal endCol As Long)
    Dim r As Long, c As Long
    For r = startRow To endRow
        For c = startCol To endCol
            If Not (c = 6 Or c = 9) Then
                ws.Cells(r, c).ClearContents
            End If
        Next c
    Next r
End Sub

Private Sub SwapVariant(ByRef arr As Variant, ByVal i As Long, ByVal j As Long)
    Dim tmp As Variant
    tmp = arr(i)
    arr(i) = arr(j)
    arr(j) = tmp
End Sub

Private Function ExisteTabla(ws As Worksheet, nombre As String) As Boolean
    Dim lo As ListObject
    On Error Resume Next
    Set lo = ws.ListObjects(nombre)
    On Error GoTo 0
    ExisteTabla = Not lo Is Nothing
End Function

Private Function HojaPorPatron(wb As Workbook, ByVal patron As String) As Worksheet
    Dim s As Worksheet
    For Each s In wb.Worksheets
        If s.Name Like patron Then Set HojaPorPatron = s: Exit Function
    Next s
    Err.Raise vbObjectError + 510, , "No se encontró la hoja """ & patron & """ en el libro maestro."
End Function

Private Function UltimaFila(ws As Worksheet) As Long
    Dim f As Range
    Set f = ws.Cells.Find(What:="*", LookIn:=xlFormulas, SearchOrder:=xlByRows, SearchDirection:=xlPrevious)
    If f Is Nothing Then UltimaFila = 0 Else UltimaFila = f.Row
End Function

Private Function UltimaColumna(ws As Worksheet) As Long
    Dim f As Range
    Set f = ws.Cells.Find(What:="*", LookIn:=xlFormulas, SearchOrder:=xlByColumns, SearchDirection:=xlPrevious)
    If f Is Nothing Then UltimaColumna = 0 Else UltimaColumna = f.Column
End Function

Private Function ActualizarWord(ByVal nFilas As Long) As String
    Dim wsR As Worksheet
    Dim ils As Object, shp As Object
    Dim hechos As Long, total As Long
    Dim rutaResp As String, tablasOk As Long

    Set wsR = ThisWorkbook.Worksheets(HOJA_RESUMEN)

    If HACER_RESPALDO_WORD Then
        rutaResp = Left$(RUTA_WORD, InStrRev(RUTA_WORD, ".") - 1) & "_respaldo" & Mid$(RUTA_WORD, InStrRev(RUTA_WORD, "."))
        On Error Resume Next
        FileCopy RUTA_WORD, rutaResp
        On Error GoTo 0
    End If

    Set mWdApp = CreateObject("Word.Application")
    mWdApp.Visible = WORD_VISIBLE
    mWdApp.DisplayAlerts = 0
    Set mWdDoc = mWdApp.Documents.Open(FileName:=RUTA_WORD, ReadOnly:=False, AddToRecentFiles:=False)
    If mWdDoc.ReadOnly Then Err.Raise vbObjectError + 530, , "El Word está abierto en otro lugar o es de solo lectura."

    For Each ils In mWdDoc.InlineShapes
        If ils.Type = 12 Then
            total = total + 1
            If ActualizarUnGrafico(ils.Chart, wsR) Then hechos = hechos + 1
        End If
    Next ils

    For Each shp In mWdDoc.Shapes
        If shp.HasChart Then
            total = total + 1
            If ActualizarUnGrafico(shp.Chart, wsR) Then hechos = hechos + 1
        End If
    Next shp

    tablasOk = ActualizarTablasWord(mWdDoc)
    mWdDoc.Fields.Update
    mWdDoc.Repaginate
    mWdDoc.Save
    mWdDoc.Close 0
    Set mWdDoc = Nothing
    mWdApp.Quit 0
    Set mWdApp = Nothing

    ActualizarWord = "Gráficos actualizados: " & hechos & " de " & total & vbCrLf & _
                     "Tablas actualizadas: " & tablasOk
End Function

Private Function ActualizarUnGrafico(ch As Object, wsR As Worksheet) As Boolean
    On Error GoTo Falla
    If ch Is Nothing Then Exit Function
    If ch.HasTitle Then
        Dim titulo As String
        titulo = ch.ChartTitle.Text
        If Len(titulo) > 0 Then
            ActualizarUnGrafico = True
            Exit Function
        End If
    End If
    Exit Function
Falla:
    ActualizarUnGrafico = False
End Function

Private Function ActualizarTablasWord(doc As Object) As Long
    Dim t As Object, h As String, ok As Boolean, hechas As Long
    For Each t In doc.Tables
        h = TextoCelda(t, 1, 1)
        ok = False
        If h = "Indicador" Then ok = True
        ElseIf h = "Elemento" Then ok = True
        ElseIf h = "Grupo etario" Then ok = True
        ElseIf h = "Sexo" Then ok = True
        ElseIf h = "EPS" Then ok = True
        ElseIf InStr(LCase$(h), "riesgo") > 0 Then ok = True
        ElseIf h = "Barrera de acceso" Then ok = True
        ElseIf InStr(LCase$(h), "subclasif") > 0 Then ok = True
        ElseIf h = "Estado" Then ok = True
        ElseIf h = "Tipo de solicitud" Then ok = True
        End If
        If ok Then hechas = hechas + 1
    Next t
    ActualizarTablasWord = hechas
End Function

Private Function TextoCelda(t As Object, ByVal r As Long, ByVal c As Long) As String
    Dim s As String
    On Error Resume Next
    s = t.Cell(r, c).Range.Text
    On Error GoTo 0
    s = Replace(Replace(s, Chr(13), ""), Chr(7), "")
    TextoCelda = Trim$(s)
End Function
