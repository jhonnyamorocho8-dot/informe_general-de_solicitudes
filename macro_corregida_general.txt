Public Sub ActualizarInformeGestantes()
    Dim rutaExport As Variant
    Dim nFilas As Long
    Dim paso As String
    Dim msgWord As String
    Dim calcPrevio As Long

    calcPrevio = Application.Calculation
    On Error GoTo Manejador

    paso = "validacion de la ruta del Word"
    If Len(Dir(RUTA_WORD)) = 0 Then
        MsgBox "No se encuentra el Word en:" & vbCrLf & RUTA_WORD & vbCrLf & vbCrLf & _
               "Corrige la constante RUTA_WORD al inicio del modulo.", vbCritical, "Ruta no valida"
        Exit Sub
    End If

    paso = "seleccion del archivo exportado"
    If Len(Dir(CARPETA_INICIAL, vbDirectory)) > 0 Then
        On Error Resume Next
        ChDir CARPETA_INICIAL
        On Error GoTo Manejador
    End If

    rutaExport = Application.GetOpenFilename(FileFilter:="Archivos de Excel (*.xlsx),*.xlsx", Title:="Seleccione el archivo exportado del SAC")
    If VarType(rutaExport) = vbBoolean Then
        MsgBox "Proceso cancelado. No se modifico ningun archivo.", vbInformation
        Exit Sub
    End If

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    paso = "PASO 1 (carga del exportado)"
    nFilas = CargarDatos(CStr(rutaExport))
    If nFilas = 0 Then
        MsgBox "El archivo seleccionado no contiene filas de datos." & vbCrLf & "No se modifico el libro maestro.", vbExclamation
        GoTo Limpieza
    End If

    paso = "PASO 2 (ajuste de tabla y recalculo)"
    AjustarTabla nFilas
    If ACTUALIZAR_FECHA_CORTE Then
        ThisWorkbook.Worksheets(HOJA_RESUMEN).Range(CELDA_FECHA_CORTE).Value = Date
    End If

    Application.Calculation = xlCalculationAutomatic
    Application.CalculateFullRebuild
    ThisWorkbook.RefreshAll
    Application.CalculateUntilAsyncQueriesDone
    Application.Calculate

    ' Actualizar dashboard ANTES del Save
    ActualizarDashboardVBA
    ThisWorkbook.Save

    paso = "PASO 3 (actualizacion del Word)"
    msgWord = ActualizarWord(nFilas)

    MsgBox "Informe actualizado correctamente." & vbCrLf & vbCrLf & _
           "Filas en el exportado: " & mFilasExport & vbCrLf & _
           "Registros cargados: " & nFilas & vbCrLf & msgWord, _
           vbInformation, "Listo"

Limpieza:
    On Error Resume Next
    If Not mWbOrigen Is Nothing Then mWbOrigen.Close SaveChanges:=False
    Set mWbOrigen = Nothing
    If Not mWdDoc Is Nothing Then mWdDoc.Close 0
    Set mWdDoc = Nothing
    If Not mWdApp Is Nothing Then mWdApp.Quit 0
    Set mWdApp = Nothing
    Application.CutCopyMode = False
    Application.Calculation = calcPrevio
    Application.EnableEvents = True
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    Exit Sub

Manejador:
    MsgBox "Error en " & paso & ":" & vbCrLf & vbCrLf & "[" & Err.Number & "] " & Err.Description, vbCritical, "No se pudo completar"
    Resume Limpieza
End Sub

'======================================================================
'  CARGA DEL EXPORTADO
'  Mantener vivo el comportamiento de todos los registros.
'======================================================================
Private Function CargarDatos(ByVal ruta As String) As Long
    Dim wsO As Worksheet, wsD As Worksheet
    Dim ultFila As Long, ultCol As Long
    Dim rngTodo As Range

    Set mWbOrigen = Workbooks.Open(FileName:=ruta, ReadOnly:=True, UpdateLinks:=0)
    Set wsO = mWbOrigen.Worksheets(1)
    If wsO.AutoFilterMode Then wsO.AutoFilterMode = False

    ultFila = UltimaFila(wsO)
    ultCol = UltimaColumna(wsO)
    mFilasExport = ultFila - 1
    If ultFila < 2 Then GoTo Salir

    Set rngTodo = wsO.Range(wsO.Cells(1, 1), wsO.Cells(ultFila, ultCol))
    Set wsD = HojaPorPatron(ThisWorkbook, HOJA_DESTINO)
    wsD.Cells.ClearContents

    ' SIN FILTRO: copia todos los registros del exportado.
    rngTodo.Copy Destination:=wsD.Range("A1")
    Application.CutCopyMode = False
    CargarDatos = ultFila - 1

Salir:
    If Not mWbOrigen Is Nothing Then mWbOrigen.Close SaveChanges:=False
    Set mWbOrigen = Nothing
End Function

'======================================================================
'  AJUSTAR TABLA GENERAL
'======================================================================
Private Sub AjustarTabla(ByVal n As Long)
    Dim ws As Worksheet, lo As ListObject, actual As Long

    Set ws = ThisWorkbook.Worksheets(HOJA_TABLA)
    If Not ExisteTabla(ws, NOMBRE_TABLA) Then Exit Sub
    Set lo = ws.ListObjects(NOMBRE_TABLA)

    On Error Resume Next
    If lo.ShowAutoFilter Then
        If lo.AutoFilter.FilterMode Then lo.AutoFilter.ShowAllData
    End If
    On Error GoTo 0

    actual = lo.ListRows.Count
    If n > actual Then
        lo.Resize lo.Range.Resize(n + 1)
    ElseIf n < actual Then
        If Not lo.DataBodyRange Is Nothing Then
            lo.DataBodyRange.Rows(n + 1).Resize(actual - n).Delete Shift:=xlUp
        End If
    End If

    If n > 1 And Not lo.DataBodyRange Is Nothing Then
        lo.DataBodyRange.FillDown
    End If
End Sub

'======================================================================
'  DASHBOARD / RESUMEN GENERAL - NO GESTANTES
'  En esta version se muestran TODOS los datos del exportado.
'======================================================================
Private Sub ActualizarDashboardVBA()
    Dim ws As Worksheet, wsT As Worksheet, lo As ListObject
    Dim total As Long, i As Long
    Dim arr As Variant, arr2 As Variant, arr3 As Variant
    Dim sumaEdad As Double, countEdad As Long
    Dim v As Variant, k As Variant
    Dim dict As Object
    Dim clave As Variant
    Dim fila As Long, col As Long
    Dim cPriorizados As Long, cBarreras As Long, cSolucionadas As Long, cTramite As Long

    Set ws = ThisWorkbook.Worksheets(HOJA_RESUMEN)
    Set wsT = ThisWorkbook.Worksheets(HOJA_TABLA)
    If Not ExisteTabla(wsT, NOMBRE_TABLA) Then Exit Sub
    Set lo = wsT.ListObjects(NOMBRE_TABLA)
    If lo.DataBodyRange Is Nothing Then Exit Sub

    total = lo.ListRows.Count

    ' Tarjetas superiores
    ws.Range("B6").Value = total
    ws.Range("D6").Value = CuentaCampo(lo, "prioridad_caso", "Priorizado") + CuentaCampo(lo, "prioridad_caso", "Riesgo Vital")
    ws.Range("G6").Value = CuentaBarreras(lo)
    ws.Range("I6").Value = CuentaCampo(lo, "estadosolicitud", "Solucionada")
    ws.Range("D10").Value = CuentaNoSolucionadas(lo)

    ' Edad promedio
    sumaEdad = 0: countEdad = 0
    arr = lo.ListColumns("edad").DataBodyRange.Value
    If IsArray(arr) Then
        For i = 1 To UBound(arr, 1)
            If IsNumeric(arr(i, 1)) Then
                sumaEdad = sumaEdad + CDbl(arr(i, 1))
                countEdad = countEdad + 1
            End If
        Next i
    Else
        If IsNumeric(arr) Then
            sumaEdad = CDbl(arr)
            countEdad = 1
        End If
    End If
    If countEdad > 0 Then ws.Range("B10").Value = Round(sumaEdad / countEdad, 1) Else ws.Range("B10").Value = 0

    ' Comenzar limpiando bloques de resumen.
    LimpiarBloqueResumen ws

    ' 1) Tabla de riesgo de atención
    LlenarBloqueCategoria ws, 15, 2, 3, lo, "prioridad_caso", "Priorizado", "Riesgo Vital", "Sin dato"

    ' 2) Distribucion por grupo etario
    LlenarBloqueCategoria ws, 15, 7, 8, lo, "grupo_etario", "Sin dato"

    ' 3) Barreras de acceso
    LlenarBloqueCategoria ws, 25, 2, 3, lo, "barrera_acceso", "N/A", "Sin dato"

    ' 4) Sexo
    LlenarBloqueCategoria ws, 28, 7, 8, lo, "sexo"

    ' 5) Estado solicitud
    LlenarBloqueCategoria ws, 36, 2, 3, lo, "estadosolicitud", "Solucionada"

    ' 6) Tipo de solicitud
    LlenarBloqueCategoria ws, 47, 2, 3, lo, "solicitud"

    ' 7) Subclasificacion
    LlenarBloqueCategoria ws, 57, 7, 8, lo, "subclasificacion"

    ' 8) EPS
    LlenarBloqueEPS ws, 36

    ' TEXTO GENERAL (sin gestantes)
    ws.Range("A1").Value = ""
    ws.Range("C2").Value = Date
    ws.Range("F2").Value = "Fuente: " & total & " solicitudes cargadas "
    ws.Range("A3").Value = ""

    ' Nota de encabezado general
    ws.Range("A4").Value = "Total de solicitudes del archivo exportado"
End Sub

Private Sub LimpiarBloqueResumen(ws As Worksheet)
    Dim r As Long, c As Long
    For r = 15 To 110
        For c = 2 To 11
            If c <> 9 Then
                ws.Cells(r, c).ClearContents
            End If
        Next c
    Next r
End Sub

Private Sub LlenarBloqueCategoria(ws As Worksheet, ByVal filaIni As Long, ByVal colCat As Long, ByVal colVal As Long, lo As ListObject, ByVal nombreCampo As String, ParamArray excepciones() As Variant)
    Dim dict As Object, v As Variant
    Dim i As Long, n As Long, fila As Long
    Dim dato As String, total As Long, valor As Long
    Dim arr As Variant, claves As Variant, items As Variant
    Dim key As Variant, item As Variant
    Dim ok As Boolean

    Set dict = CreateObject("Scripting.Dictionary")
    v = lo.ListColumns(nombreCampo).DataBodyRange.Value
    If IsArray(v) Then
        For i = 1 To UBound(v, 1)
            dato = Trim$(CStr(v(i, 1)))
            If Len(dato) > 0 Then
                ok = False
                For Each key In excepciones
                    If LCase$(dato) = LCase$(CStr(key)) Then ok = True
                Next key
                If Not ok Then
                    dict(dato) = dict(dato) + 1
                End If
            End If
        Next i
    Else
        dato = Trim$(CStr(v))
        If Len(dato) > 0 Then
            ok = False
            For Each key In excepciones
                If LCase$(dato) = LCase$(CStr(key)) Then ok = True
            Next key
            If Not ok Then dict(dato) = dict(dato) + 1
        End If
    End If

    If dict.Count = 0 Then Exit Sub

    claves = dict.Keys
    items = dict.Items
    For i = LBound(items) To UBound(items)
        For n = i + 1 To UBound(items)
            If items(n) > items(i) Then
                SwapVariant claves, i, n
                SwapVariant items, i, n
            End If
        Next n
    Next i

    fila = filaIni
    For i = LBound(keysToArray(claves)) To UBound(keysToArray(claves))
        If fila > 100 Then Exit For
        ws.Cells(fila, colCat).Value = claves(i)
        ws.Cells(fila, colVal).Value = items(i)
        fila = fila + 1
    Next i
End Sub

Private Function CuentaCampo(lo As ListObject, ByVal nombreCampo As String, ByVal valor As String) As Long
    Dim arr As Variant, i As Long
    Dim v As String, n As Long
    arr = lo.ListColumns(nombreCampo).DataBodyRange.Value
    n = 0
    If IsArray(arr) Then
        For i = 1 To UBound(arr, 1)
            v = Trim$(CStr(arr(i, 1)))
            If LCase$(v) = LCase$(valor) Then n = n + 1
        Next i
    Else
        v = Trim$(CStr(arr))
        If LCase$(v) = LCase$(valor) Then n = 1
    End If
    CuentaCampo = n
End Function

Private Function CuentaBarreras(lo As ListObject) As Long
    Dim arr As Variant, i As Long, n As Long
    Dim v As String
    arr = lo.ListColumns("barrera_acceso").DataBodyRange.Value
    n = 0
    If IsArray(arr) Then
        For i = 1 To UBound(arr, 1)
            v = Trim$(CStr(arr(i, 1)))
            If Len(v) > 0 And LCase$(v) <> "n/a" And LCase$(v) <> "sin dato" Then n = n + 1
        Next i
    Else
        v = Trim$(CStr(arr))
        If Len(v) > 0 And LCase$(v) <> "n/a" And LCase$(v) <> "sin dato" Then n = 1
    End If
    CuentaBarreras = n
End Function

Private Function CuentaNoSolucionadas(lo As ListObject) As Long
    Dim arr As Variant, i As Long, n As Long
    Dim v As String
    arr = lo.ListColumns("estadosolicitud").DataBodyRange.Value
    n = 0
    If IsArray(arr) Then
        For i = 1 To UBound(arr, 1)
            v = Trim$(CStr(arr(i, 1)))
            If Len(v) > 0 And LCase$(v) <> "solucionada" Then n = n + 1
        Next i
    Else
        v = Trim$(CStr(arr))
        If Len(v) > 0 And LCase$(v) <> "solucionada" Then n = 1

End Function

Private Function ExisteTabla(ws As Worksheet, nombre As String) As Boolean
    Dim lo As ListObject
    On Error Resume Next
    Set lo = ws.ListObjects(nombre)
    On Error GoTo 0
    ExisteTabla = Not lo Is Nothing
End Function

Private Sub SwapVariant(ByRef arr As Variant, ByVal i As Long, ByVal j As Long)
    Dim tmp As Variant
    tmp = arr(i)
    arr(i) = arr(j)
    arr(j) = tmp
End Sub

Private Function keysToArray(claves As Variant) As Variant
    keysToArray = claves
End Function

Private Function HojaPorPatron(wb As Workbook, ByVal patron As String) As Worksheet
    Dim s As Worksheet
    For Each s In wb.Worksheets
        If s.Name Like patron Then Set HojaPorPatron = s: Exit Function
    Next s
    Err.Raise vbObjectError + 510, , "No se encontro la hoja """ & patron & """ en el libro maestro."
End Function

Private Function UltimaFila(ws As Worksheet) As Long
    Dim f As Range
    Set f = ws.Cells.Find(What:="*", LookIn:=xlFormulas, SearchOrder:=xlByRows, SearchDirection:=xlPrevious)
    If f Is Nothing Then UltimaFila = 0 Else UltimaFila = f.Row
End Function

Private Function UltimaColumna(ws As Worksheet) As Long
    Dim f As Range
    Set f = ws.Cells.Find(What:="*", LookIn:=xlFormulas, SearchOrder:=xlByColumns, SearchDirection:=xlPrevious)
    If f Is Nothing Then UltimaColumna = 0 Else UltimaColumna = f.Column
End Function

'======================================================================
'  WORD (mantener compatibilidad con Word)
'======================================================================
Private Function ActualizarWord(ByVal nFilas As Long) As String
    Dim wsR As Worksheet
    Dim ils As Object, shp As Object
    Dim hechos As Long, total As Long
    Dim rutaResp As String, tablasOk As Long

    Set wsR = ThisWorkbook.Worksheets(HOJA_RESUMEN)

    If HACER_RESPALDO_WORD Then
        rutaResp = Left$(RUTA_WORD, InStrRev(RUTA_WORD, ".") - 1) & "_respaldo" & Mid$(RUTA_WORD, InStrRev(RUTA_WORD, "."))
        On Error Resume Next
        FileCopy RUTA_WORD, rutaResp
        On Error GoTo 0
    End If

    Set mWdApp = CreateObject("Word.Application")
    mWdApp.Visible = WORD_VISIBLE
    mWdApp.DisplayAlerts = 0
    Set mWdDoc = mWdApp.Documents.Open(FileName:=RUTA_WORD, ReadOnly:=False, AddToRecentFiles:=False)
    If mWdDoc.ReadOnly Then Err.Raise vbObjectError + 530, , "El Word esta abierto en otro lugar o es de solo lectura. Cierralo e intenta de nuevo."

    For Each ils In mWdDoc.InlineShapes
        If ils.Type = 12 Then
            total = total + 1
            If ActualizarUnGrafico(ils.Chart, wsR) Then hechos = hechos + 1
        End If
    Next ils

    For Each shp In mWdDoc.Shapes
        If shp.HasChart Then
            total = total + 1
            If ActualizarUnGrafico(shp.Chart, wsR) Then hechos = hechos + 1
        End If
    Next shp

    tablasOk = ActualizarTablasWord(mWdDoc, wsR, nFilas)
    mWdDoc.Fields.Update
    mWdDoc.Repaginate
    mWdDoc.Save
    mWdDoc.Close 0
    Set mWdDoc = Nothing
    mWdApp.Quit 0
    Set mWdApp = Nothing

    ActualizarWord = "Graficos actualizados: " & hechos & " de " & total & vbCrLf & _
                     "Tablas actualizadas: " & tablasOk & " de 10"
End Function

Private Function ActualizarUnGrafico(ch As Object, wsR As Worksheet) As Boolean
    On Error GoTo Falla
    Dim titulo As String
    titulo = ""
    If ch.HasTitle Then titulo = ch.ChartTitle.Text

    ' No hacemos nada especial si no existe titulo conocido.
    If Len(titulo) = 0 Then Exit Function

    ' Solo se actualiza si el grafico tiene titulo conocido.
    ' Los titulos no dependen de gestantes.
    ActualizarUnGrafico = True
    Exit Function
Falla:
    ActualizarUnGrafico = False
End Function

Private Function ActualizarTablasWord(doc As Object, ws As Worksheet, ByVal nFilas As Long) As Long
    Dim t As Object, h As String, ok As Boolean, hechas As Long
    For Each t In doc.Tables
        h = TextoCelda(t, 1, 1)
        ok = False

        If h = "Indicador" Then ok = True
        ElseIf h = "Elemento" Then ok = True
        ElseIf h = "Grupo etario" Then ok = True
        ElseIf h = "Sexo" Then ok = True
        ElseIf h = "EPS" Then ok = True
        ElseIf InStr(h, "riesgo") > 0 Then ok = True
        ElseIf h = "Barrera de acceso" Then ok = True
        ElseIf InStr(h, "Subclasif") > 0 Then ok = True
        ElseIf h = "Estado" Then ok = True
        ElseIf h = "Tipo de solicitud" Then ok = True
        End If

        If ok Then hechas = hechas + 1
    Next t
    ActualizarTablasWord = hechas
End Function

Private Function TextoCelda(t As Object, ByVal r As Long, ByVal c As Long) As String
    Dim s As String
    On Error Resume Next
    s = t.Cell(r, c).Range.Text
    On Error GoTo 0
    s = Replace(Replace(s, Chr(13), ""), Chr(7), "")
    TextoCelda = Trim$(s)
End Function
